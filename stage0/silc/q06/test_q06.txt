class
A {
    decl
        final int f();
        int g();
    enddecl
    int f() {
        decl enddecl
        begin
            return 1;
        end
    }
    int g() {
        decl enddecl
        begin
            return 2;
        end
    }
}
B extends A {
    decl
        int g();
    enddecl
    int g() {
        decl enddecl
        begin
            return 20;
        end
    }
}
endclass
decl
    A o;
enddecl
int main() {
    decl enddecl
    begin
        initialize();
        o = new(B);
        write(o.f());
        write(o.g());
        return 0;
    end
}
