int main() {
    decl int i, s; enddecl
    begin
        s = 0;
        for (i = 0; i < 10; i = i + 1) do
            if (i == 2) then
                continue;
            endif;
            if (i == 6) then
                break;
            endif;
            s = s + i;
        endfor;
        write(s);
        write(i);
        return 0;
    end
}
