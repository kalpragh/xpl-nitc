class
Shape {
    decl
        int readDims();
        int area();
    enddecl
    int readDims() {
        decl enddecl
        begin
            return 0;
        end
    }
    int area() {
        decl enddecl
        begin
            return 0;
        end
    }
}
Rectangle extends Shape {
    decl
        int length;
        int breadth;
        int readDims();
        int area();
    enddecl
    int readDims() {
        decl enddecl
        begin
            read(self.length);
            read(self.breadth);
            return 0;
        end
    }
    int area() {
        decl enddecl
        begin
            return self.length * self.breadth;
        end
    }
}
Square extends Shape {
    decl
        int side;
        int readDims();
        int area();
    enddecl
    int readDims() {
        decl enddecl
        begin
            read(self.side);
            return 0;
        end
    }
    int area() {
        decl enddecl
        begin
            return self.side * self.side;
        end
    }
}
endclass
decl
    Shape s;
enddecl
int main() {
    decl
        int choice, t, r;
    enddecl
    begin
        initialize();
        read(choice);
        if (choice == 1) then
            s = new(Rectangle);
        else
            s = new(Square);
        endif;
        t = s.readDims();
        r = s.area();
        write(r);
        return 0;
    end
}
