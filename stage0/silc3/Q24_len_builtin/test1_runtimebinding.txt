class
A {
    decl
        int F1();
    enddecl
    int F1() {
        decl enddecl
        begin
            write("In A F1");
            return 0;
        end
    }
}
B extends A {
    decl
        int F1();
    enddecl
    int F1() {
        decl enddecl
        begin
            write("In B F1");
            return 1;
        end
    }
}
endclass
decl
    A x;
enddecl
int main() {
    decl
        int n, retval;
    enddecl
    begin
        initialize();
        read(n);
        if (n > 0) then
            x = new(A);
        else
            x = new(B);
        endif;
        retval = x.F1();
        write(retval);
        return 0;
    end
}
