decl
    int a, b;
    int swap(int *p, int *q);
    int inc(int *p);
    int twice(int *p);
    int main();
enddecl
int swap(int *p, int *q) {
    decl
        int t;
    enddecl
    begin
        t = *p;
        *p = *q;
        *q = t;
        return 0;
    end
}
int inc(int *p) {
    decl enddecl
    begin
        *p = *p + 1;
        return 0;
    end
}
int twice(int *p) {
    decl
        int d;
    enddecl
    begin
        d = inc(p);
        d = inc(p);
        return 0;
    end
}
