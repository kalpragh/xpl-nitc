class
  Bst {
    decl
      int key;
      Bst left;
      Bst right;
      int init(int k);
      int insert(int k);
      int inorder();
      int search(int k);
    enddecl
    int init(int k){
      decl enddecl
      begin
        self.key = k;
        self.left = NULL;
        self.right = NULL;
        return 1;
      end
    }
    int insert(int k){
      decl int t; enddecl
      begin
        if (k < self.key) then
          if (self.left == NULL) then
            self.left = new(Bst);
            t = self.left.init(k);
          else
            t = self.left.insert(k);
          endif;
        else
          if (self.right == NULL) then
            self.right = new(Bst);
            t = self.right.init(k);
          else
            t = self.right.insert(k);
          endif;
        endif;
        return 1;
      end
    }
    int inorder(){
      decl int t; enddecl
      begin
        if (self.left != NULL) then
          t = self.left.inorder();
        endif;
        write(self.key);
        if (self.right != NULL) then
          t = self.right.inorder();
        endif;
        return 1;
      end
    }
    int search(int k){
      decl int r; enddecl
      begin
        r = 0;
        if (k == self.key) then
          r = 1;
        else
          if (k < self.key) then
            if (self.left != NULL) then
              r = self.left.search(k);
            endif;
          else
            if (self.right != NULL) then
              r = self.right.search(k);
            endif;
          endif;
        endif;
        return r;
      end
    }
  }
endclass
decl Bst root; enddecl
int main(){
  decl int n; int i; int x; int t; enddecl
  begin
    initialize();
    read(n);
    read(x);
    root = new(Bst);
    t = root.init(x);
    i = 1;
    while (i < n) do
      read(x);
      t = root.insert(x);
      i = i + 1;
    endwhile;
    t = root.inorder();
    read(x);
    write(root.search(x));
    read(x);
    write(root.search(x));
    return 0;
  end
}
